cmake_minimum_required(VERSION 3.18)
project(gpt2codex_gui LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)

option(G2C_BUILD_QTERMWIDGET "Build optional QTermWidget terminal backend" ON)
option(G2C_BUILD_KONSOLE "Build optional Konsole KPart terminal backend" ON)
option(G2C_REQUIRE_ALL_BACKENDS "Fail packaging if either requested backend is missing" OFF)
if(G2C_BUILD_QTERMWIDGET)
    find_package(qtermwidget6 CONFIG QUIET)
endif()
if(G2C_BUILD_KONSOLE)
    find_package(KF6Parts CONFIG QUIET)
    find_package(KF6CoreAddons CONFIG QUIET)
endif()

add_executable(gpt2codex-gui main.cpp)
target_link_libraries(gpt2codex-gui PRIVATE Qt6::Widgets)
target_compile_options(gpt2codex-gui PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(gpt2codex-gui PRIVATE "G2C_INSTALLED_BACKEND_DIR=\"${CMAKE_INSTALL_FULL_LIBDIR}/gpt2codex/backends\"")
install(TARGETS gpt2codex-gui RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")

if(CMAKE_RUNTIME_OUTPUT_DIRECTORY)
    set(G2C_BACKEND_BUILD_DIR "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/gpt2codex-backends")
else()
    set(G2C_BACKEND_BUILD_DIR "${CMAKE_CURRENT_BINARY_DIR}/gpt2codex-backends")
endif()
set(G2C_BACKEND_INSTALL_DIR "${CMAKE_INSTALL_LIBDIR}/gpt2codex/backends")
set(G2C_BACKEND_COUNT 0)
set(G2C_HAVE_QTERM false)
set(G2C_HAVE_KONSOLE false)

if(G2C_BUILD_QTERMWIDGET AND qtermwidget6_FOUND)
    # Debian 13's QTermWidget CMake target requires the unversioned
    # libutf8proc.so linker library from libutf8proc-dev.
    find_library(UTF8PROC_LINK_LIBRARY NAMES utf8proc)
    if(NOT UTF8PROC_LINK_LIBRARY)
        message(FATAL_ERROR "QTermWidget requires libutf8proc-dev for linking")
    endif()
    add_library(gpt2codex-backend-qtermwidget MODULE backend_qtermwidget.cpp)
    target_link_libraries(gpt2codex-backend-qtermwidget PRIVATE Qt6::Widgets qtermwidget6)
    target_compile_options(gpt2codex-backend-qtermwidget PRIVATE -Wall -Wextra -Wpedantic)
    set_target_properties(gpt2codex-backend-qtermwidget PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${G2C_BACKEND_BUILD_DIR}")
    install(TARGETS gpt2codex-backend-qtermwidget LIBRARY DESTINATION "${G2C_BACKEND_INSTALL_DIR}")
    math(EXPR G2C_BACKEND_COUNT "${G2C_BACKEND_COUNT}+1")
    set(G2C_HAVE_QTERM true)
    message(STATUS "gpt2codex: QTermWidget6 backend enabled")
else()
    message(STATUS "gpt2codex: QTermWidget6 backend not built")
endif()

if(G2C_BUILD_KONSOLE AND KF6Parts_FOUND AND KF6CoreAddons_FOUND)
    add_library(gpt2codex-backend-konsole MODULE backend_konsole.cpp)
    target_link_libraries(gpt2codex-backend-konsole PRIVATE Qt6::Widgets KF6::Parts KF6::CoreAddons KF6::ConfigCore)
    target_compile_options(gpt2codex-backend-konsole PRIVATE -Wall -Wextra -Wpedantic)
    set_target_properties(gpt2codex-backend-konsole PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${G2C_BACKEND_BUILD_DIR}")
    install(TARGETS gpt2codex-backend-konsole LIBRARY DESTINATION "${G2C_BACKEND_INSTALL_DIR}")
    math(EXPR G2C_BACKEND_COUNT "${G2C_BACKEND_COUNT}+1")
    set(G2C_HAVE_KONSOLE true)
    message(STATUS "gpt2codex: Konsole KPart backend enabled (runtime requires konsole-kpart)")
else()
    message(STATUS "gpt2codex: Konsole KPart backend not built")
endif()

if(G2C_BACKEND_COUNT EQUAL 0)
    message(FATAL_ERROR "No terminal backend can be built. Install libqtermwidget-dev + libutf8proc-dev and/or libkf6parts-dev + libkf6coreaddons-dev")
endif()
if(G2C_REQUIRE_ALL_BACKENDS AND (NOT G2C_HAVE_QTERM OR NOT G2C_HAVE_KONSOLE))
    message(FATAL_ERROR "Alpha package requires both terminal modules; refusing an incomplete package")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/backends.json"
    "{\"konsole\": ${G2C_HAVE_KONSOLE}, \"qtermwidget\": ${G2C_HAVE_QTERM}}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/backends.json" DESTINATION "${CMAKE_INSTALL_DATADIR}/gpt2codex")

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
    add_executable(gpt2codex-gui-tests main.cpp lifecycle_tests.h)
    set_target_properties(gpt2codex-gui-tests PROPERTIES AUTOMOC_PATH_PREFIX ON)
    target_include_directories(gpt2codex-gui-tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(gpt2codex-gui-tests PRIVATE G2C_GUI_TEST
        "G2C_GUI_TEST_WORK_DIR=\"${CMAKE_CURRENT_BINARY_DIR}\""
        "G2C_INSTALLED_BACKEND_DIR=\"${CMAKE_INSTALL_FULL_LIBDIR}/gpt2codex/backends\"")
    target_link_libraries(gpt2codex-gui-tests PRIVATE Qt6::Widgets Qt6::Test)
    add_test(NAME gui-lifecycle-and-paste COMMAND gpt2codex-gui-tests)
    set_tests_properties(gui-lifecycle-and-paste PROPERTIES
        ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=generic")
    add_executable(gpt2codex-backend-tests backend_tests.cpp)
    target_link_libraries(gpt2codex-backend-tests PRIVATE Qt6::Widgets Qt6::Test)
    target_compile_definitions(gpt2codex-backend-tests PRIVATE
        "G2C_TEST_WORK_DIR=\"${CMAKE_CURRENT_BINARY_DIR}\"")
    if(G2C_BUILD_KONSOLE AND KF6Parts_FOUND AND KF6CoreAddons_FOUND)
        target_compile_definitions(gpt2codex-backend-tests PRIVATE G2C_TEST_KONSOLE)
        target_link_libraries(gpt2codex-backend-tests PRIVATE KF6::ConfigCore)
        add_dependencies(gpt2codex-backend-tests gpt2codex-backend-konsole)
        add_test(NAME gui-konsole-integration COMMAND gpt2codex-backend-tests konsoleShortcutsAndLayout)
        set_tests_properties(gui-konsole-integration PROPERTIES
            ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=generic" TIMEOUT 20)
    endif()
    if(G2C_BUILD_QTERMWIDGET AND qtermwidget6_FOUND)
        add_dependencies(gpt2codex-backend-tests gpt2codex-backend-qtermwidget)
        add_test(NAME gui-qterm-render COMMAND gpt2codex-backend-tests renderQTerm)
        set_tests_properties(gui-qterm-render PROPERTIES
            ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=generic" TIMEOUT 20)
    endif()
endif()
